\begin{figure}[ht]

  \centering
  \begin{tikzpicture}[
      % Define block styles
      block/.style={
          draw, 
          rectangle, 
          minimum width=2.5cm, 
          minimum height=1cm, 
          thick, 
          fill=white,
          font=\sffamily\bfseries
      },
      % Define the thick arrow style
      bus/.style={
          <->, 
          line width=4pt, 
          >={Triangle[length=8pt, width=15pt]}
      },
      % Define a one-way thick arrow
      bus_single/.style={
          ->, 
          line width=4pt, 
          >={Triangle[length=8pt, width=15pt]}
      }
  ]

      % Motherboard Outer Box
      \draw[thick, draw=red] (-5.5,-4.5) rectangle (10,3);
      \node[anchor=north west, font=\huge\sffamily] at (-4.8,2.2) {MOTHERBOARD};

      % Outside/Top
      \node[block] (cpu) at (1,4) {CPU};

      % CLOCK_CONTROLLER
      \node[block] (clock_controller) at (3.5,2) {CLOCK\textunderscore CONTROLLER};

      % Center Node
      \node[block] (mem_support) at (1,0) {MEMORY\textunderscore SUPPORT};

      % Left Nodes
      \node[block, left=1.5cm of mem_support] (memory) {MEMORY};
      \node[block, below=1.5cm of memory] (screencard) {SCREENCARD};

      % Right Nodes
      \node[block, right=1.5cm of mem_support] (eeprom) {EEPROM\textunderscore MANAGER};

      % Bottom Nodes
      \node[block, below=1.5cm of mem_support] (ps2) {PS2\textunderscore MODUL};
      \node[block, right=0.8cm of ps2] (random) {RANDOM};
      \node[block, right=0.8cm of random] (timer) {TIMER\textunderscore MILLIS};

      % Connections (Thick Arrows)
      \begin{pgfonlayer}{background}
          \draw[bus] (cpu) -- (mem_support);
          \draw[bus] (mem_support) -- (memory);
          \draw[bus] (mem_support) -- (eeprom);
          \draw[bus] (mem_support) -- (ps2);
          
          \draw[bus_single] (memory) -- (screencard);
          \draw[bus_single] (clock_controller) -- (mem_support);
          \draw[bus_single] (clock_controller) -- (cpu);
          \draw[bus_single] ($(random.center)!0.7!(random.west)$) -- ($(mem_support.south)!0.5!(mem_support.south east)$);
          % Pointing to the center ensures the box hides the messy tip
          \draw[bus_single] (timer.center) -- (mem_support.south east); 
      \end{pgfonlayer}

  \end{tikzpicture}
  \caption{Block diagram for all FPGA-modules}
  \label{fig:fpga_block_diagram_motherboard}
\end{figure}
